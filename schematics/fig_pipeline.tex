% 2D rock-salt cartoon of the pristine supercell; (\vx,\vy) is the flagged O site.
\newcommand{\cell}[1]{% #1 = 1 draws radius-graph edges
  \def\a{0.36}
  \ifnum#1=1
    \foreach \i in {0,...,4} \foreach \j in {0,...,4} {
      \foreach \k/\l in {1/0,0/1,1/1,1/-1} {
        \pgfmathtruncatemacro{\ii}{\i+\k}\pgfmathtruncatemacro{\jj}{\j+\l}
        \ifnum\ii<5 \ifnum\jj<5 \ifnum\jj>-1
          \draw[black!28, line width=0.35pt] (\i*\a,\j*\a) -- (\ii*\a,\jj*\a);
        \fi\fi\fi
      }
    }
  \fi
  \foreach \i in {0,...,4} \foreach \j in {0,...,4} {
    \pgfmathtruncatemacro{\p}{mod(\i+\j,2)}
    \ifnum\i=2 \ifnum\j=1 \def\p{2}\fi\fi
    \ifcase\p
      \node[atom=2.1mm, fill=black!55, draw=black!70, line width=0.3pt] at (\i*\a,\j*\a) {};
    \or
      \node[atom=1.6mm, fill=white, draw=black!60, line width=0.4pt] at (\i*\a,\j*\a) {};
    \or
      \node[atom=1.75mm, fill=cO, draw=cO!60!black, line width=0.5pt] at (\i*\a,\j*\a) {};
    \fi
  }
}
