  % ---------------- inputs ----------------
  \tikzset{inp/.style={quantity, minimum width=2.6cm, minimum height=5.6mm}}

  \node[inp] (xi) at (1.30, 0.40) {nodes $\mathbf{x}_i^{(0)}$};
  \node[inp] (eij) at (1.30,-0.40) {edges $\mathbf{e}_{ij}^{(0)}$};
  \node[inp] (u0) at (1.30,-1.20) {global state $u^{(0)}$};

  % D-state injection points
  \node[quantity, draw=cD, fill=cD!14] (site) at (1.30, 2.35) {site-electronic};

  \draw[inject] (host) -- (u0);

  % ---------------- outputs and pooling ----------------
  \tikzset{outp/.style={quantity, minimum height=5.6mm}}
  \node[outp] (hv) at (6.25, 1.20) {$\mathbf{h}_\mathrm{v}$};

  \node[outp] (uo) at (6.25,-1.20) {$u$};

  \node[box, draw=cS, align=center] (pool) at (7.75,0) {Set2Set\\or mean};
  \draw[flow] (hi) -- (pool);

  % ---------------- concatenated readout vector ----------------
  \tikzset{seg/.style={draw=cS, fill=cS!14, minimum width=1.25cm, minimum height=6.5mm, inner sep=0pt, line width=0.6pt}}

  \draw[flow] (hv.east) -| ($(c1.west)+(-0.30,0)$) -- (c1.west);
  \draw[flow] (pool) -- (c2);
  \draw[flow] (uo.east) -| ($(c3.west)+(-0.30,0)$) -- (c3.west);

  \node[model=cS, minimum width=1.1cm] (mlp) at (11.20,0) {MLP};

  \draw[flow] (c2.east) -- (mlp);
  \draw[flow] (mlp) -- (ef);
